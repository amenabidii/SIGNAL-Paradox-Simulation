cmake_minimum_required(VERSION 3.12)
project(SIGNAL_Observer_Paradox LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(GLOB SIGNAL_SOURCES CONFIGURE_DEPENDS "src/*.cpp")
add_executable(signal ${SIGNAL_SOURCES})
target_include_directories(signal PRIVATE include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(signal PRIVATE -Wall -Wextra -pedantic)
endif()
